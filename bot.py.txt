import logging
from telegram import Update
from telegram.ext import ApplicationBuilder, CommandHandler, MessageHandler, filters, ContextTypes
from deep_translator import GoogleTranslator

# Bot Token ကို ဒီမှာ ထည့်ပါ
TOKEN = "8631809233:AAFdyh_E9vKjs92jqGQviGCJ34wyeDNPdEo"

logging.basicConfig(
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    level=logging.INFO
)

async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text("မင်္ဂလာပါ။ မြန်မာ-စပိန် ဘာသာပြန် Bot မှ ကြိုဆိုပါတယ်။ စာများ ပို့ပေးနိုင်ပါတယ်။")

async def translate_text(update: Update, context: ContextTypes.DEFAULT_TYPE):
    text = update.message.text
    
    try:
        # စပိန်စာဖြစ်ပါက မြန်မာသို့၊ အခြားစာဖြစ်ပါက စပိန်သို့ ဘာသာပြန်ပေးမည်
        translated = GoogleTranslator(source='auto', target='es').translate(text)
        await update.message.reply_text(translated)
    except Exception as e:
        await update.message.reply_text("ဘာသာပြန်ရာတွင် အမှားအယွင်း ရှိနေပါသည်။")

if __name__ == '__main__':
    app = ApplicationBuilder().token(TOKEN).job_queue(None).build()
    
    app.add_handler(CommandHandler("start", start))
    app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, translate_text))
    
    print("Myanmar-Spanish Translate Bot အလုပ်လုပ်နေပါပြီ...")
    app.run_polling()
